# Both local backend rawValues install the same sidecar (asr_server.py), so this
# file and requirements-nemotron.txt are intentionally identical. The engine
# rawValue only selects which requirements file LocalASREnvironment installs.
#
# ASR runs nemotron-3.5-asr-streaming through sherpa-onnx (ONNX Runtime on the
# CPU). mlx-audio is gone: its streaming path could not keep up below the 1120 ms
# chunk, and the Qwen3-ASR second pass it also served has been dropped.
# Pinned exactly, on 2026-09-21, to whatever was current on PyPI that day. Two
# machines installing a week apart otherwise get different programs: mlx-lm in
# particular has changed stream_generate's chunk shape between releases, and
# the venv is built from this file on the user's machine, not shipped.
#
# To bump: pick the new versions, install them into a scratch venv on macOS
# arm64, run a real recording plus a file import through the app, then edit
# both files (they must stay identical) and bump
# LocalASREnvironment.requirementsGeneration so existing venvs reinstall.
# Transitive dependencies are deliberately left unpinned -- pinning them
# without --require-hashes buys nothing, and pip refuses a partially hashed
# file.
sherpa-onnx==1.13.8
# Fetches the ONNX weights from Hugging Face into ~/.cache/huggingface.
huggingface_hub==1.32.0
# numpy stays a minor behind: 2.5.x declares Requires-Python >=3.12, and
# LocalASREnvironment accepts any interpreter from 3.11 up. 2.4.6 is the last
# release that installs on 3.11, which is also what CI runs. Raise this only
# together with the floor in LocalASREnvironment.minimumPythonVersion.
numpy==2.4.6
websockets==17.1
# Drives translate_server.py (Hy-MT2) and llm_server.py (Qwen3); both share this
# venv with the ASR sidecar.
mlx==0.32.2
mlx-lm==0.31.3
